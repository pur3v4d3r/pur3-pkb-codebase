```python
async def smart_consultation_router(
    query: str,
    query_classification: QueryClassification,
    force_refresh: bool = False
) -> ConsultationResult:
    """
    Route query through intelligent consultation layers.
    
    NEW in v5.1.0: Dramatically reduces latency while maintaining quality.
    """
    
    if force_refresh:
        # User explicitly requests fresh search
        return await full_project_search(query)
    
    # Layer 1: Check high-confidence cache
    cache_result = await check_consultation_cache(
        query=query,
        confidence_threshold=0.90
    )
    
    if cache_result and cache_result.confidence >= 0.90:
        return ConsultationResult(
            source='cache',
            knowledge=cache_result.knowledge,
            confidence=cache_result.confidence,
            latency_ms=cache_result.latency,
            sources=cache_result.sources,
            cached=True,
            cache_age_seconds=cache_result.age
        )
    
    # Layer 2: Check semantic pattern index
    pattern_result = await check_pattern_index(
        query=query,
        similarity_threshold=0.92
    )
    
    if pattern_result and pattern_result.similarity >= 0.92:
        return ConsultationResult(
            source='pattern_match',
            knowledge=pattern_result.knowledge,
            confidence=pattern_result.confidence,
            latency_ms=pattern_result.latency,
            sources=pattern_result.sources,
            cached=False,
            pattern_similarity=pattern_result.similarity
        )
    
    # Layer 3: Full project knowledge search
    search_result = await full_project_search(
        query=query,
        max_results=8
    )
    
    # Cache result for future queries if high confidence
    if search_result.confidence >= 0.85:
        await cache_consultation_result(
            query=query,
            result=search_result,
            ttl=3600
        )
    
    # Update pattern index if novel query
    if await is_novel_query(query):
        await update_pattern_index(
            query=query,
            result=search_result
        )
    
    return ConsultationResult(
        source='full_search',
        knowledge=search_result.knowledge,
        confidence=search_result.confidence,
        latency_ms=search_result.latency,
        sources=search_result.sources,
        cached=False
```

## Consultation Confidence Scoring

[**Confidence-Scoring-System**:: Multi-factor scoring system that evaluates the reliability and applicability of consultation results based on semantic relevance, source authority, recency, and query-result alignment.]

### Confidence Calculation

```python
class ConsultationConfidenceScorer:
    """
    Calculate confidence scores for consultation results.
    
    NEW in v5.1.0: Enables intelligent cache routing decisions.
    """
    
    def __init__(self):
        self.factor_weights = {
            'semantic_relevance': 0.35,
            'source_authority': 0.25,
            'recency': 0.15,
            'query_result_alignment': 0.20,
            'internal_consistency': 0.05
        }
    
    def calculate_confidence(
        self,
        query: str,
        result: ConsultationResult,
        context: Dict[str, Any]
    ) -> float:
        """
        Calculate overall confidence score (0-1).
        """
        
        scores = {}
        
        # Factor 1: Semantic Relevance
        scores['semantic_relevance'] = self._score_semantic_relevance(
            query, result.sources
        )
        
        # Factor 2: Source Authority
        scores['source_authority'] = self._score_source_authority(
            result.sources
        )
        
        # Factor 3: Recency
        scores['recency'] = self._score_recency(
            result.sources
        )
        
        # Factor 4: Query-Result Alignment
        scores['query_result_alignment'] = self._score_alignment(
            query, result.knowledge
        )
        
        # Factor 5: Internal Consistency
        scores['internal_consistency'] = self._score_consistency(
            result.sources
        )
        
        # Calculate weighted total
        confidence = sum(
            scores[factor] * self.factor_weights[factor]
            for factor in scores
        )
        
        return confidence
    
    def _score_semantic_relevance(
        self,
        query: str,
        sources: List[KnowledgeSource]
    ) -> float:
        """
        Score how semantically relevant sources are to query.
        """
        if not sources:
            return 0.0
        
        relevance_scores = []
        for source in sources:
            # Compute semantic similarity between query and source
            similarity = self._compute_similarity(
                query,
                source.content
            )
            relevance_scores.append(similarity)
        
        # Use weighted average (higher weight to top sources)
        weights = [1.0 / (i + 1) for i in range(len(relevance_scores))]
        weighted_score = sum(
            s * w for s, w in zip(relevance_scores, weights)
        ) / sum(weights)
        
        return weighted_score
    
    def _score_source_authority(
        self,
        sources: List[KnowledgeSource]
    ) -> float:
        """
        Score authority level of sources.
        """
        authority_levels = {
            'doc1': 0.95,  # Core manuals have highest authority
            'doc2': 0.95,
            'doc3': 0.95,
            'doc4': 0.95,
            'doc5': 0.95,
            'doc6': 0.95,
            '01-guide': 0.90,  # Guides have high authority
            '02-guide': 0.90,
            '03-guide': 0.90,
            '04-guide': 0.90,
            '05-guide': 0.90,
            '06-guide': 0.90,
            'templates': 0.85,  # Templates have good authority
            'metadata': 0.85,
            'examples': 0.80,  # Examples have moderate authority
        }
        
        if not sources:
            return 0.0
        
        scores = []
        for source in sources:
            # Extract document type from source
            doc_type = self._extract_doc_type(source.document_id)
            authority = authority_levels.get(doc_type, 0.70)
            scores.append(authority)
        
        return sum(scores) / len(scores)
    
    def _score_recency(
        self,
        sources: List[KnowledgeSource]
    ) -> float:
        """
        Score recency of sources.
        """
        if not sources:
            return 0.0
        
        current_time = datetime.utcnow()
        recency_scores = []
        
        for source in sources:
            # Get last modified date
            last_modified = source.metadata.get('last_modified')
            if not last_modified:
                recency_scores.append(0.70)  # Unknown age, moderate score
                continue
            
            # Calculate age in days
            age_days = (current_time - last_modified).days
            
            # Score based on age (exponential decay)
            if age_days <= 7:
                score = 1.0
            elif age_days <= 30:
                score = 0.95
            elif age_days <= 90:
                score = 0.90
            elif age_days <= 180:
                score = 0.85
            else:
                score = 0.80
            
            recency_scores.append(score)
        
        return sum(recency_scores) / len(recency_scores)
```

```python
class AutomatedCitationValidator:
    """
    Automatically validate all citations against project knowledge sources.
    
    NEW in v5.1.0: Ensures citation integrity and prevents fabrication.
    """
    
    def __init__(self):
        self.document_cache = {}
        self.validation_history = []
        self.violation_threshold = 0.85  # Semantic similarity threshold
    
    async def validate_response_citations(
        self,
        response: str,
        consultation_context: ConsultationContext
    ) -> ValidationReport:
        """
        Validate all citations in response against sources.
        """
        
        # Extract all citations from response
        citations = self._extract_citations(response)
        
        if not citations:
            return ValidationReport(
                status='NO_CITATIONS',
                total_citations=0,
                verified=0,
                failed=0,
                warnings=[]
            )
        
        validation_results = []
        
        for citation in citations:
            result = await self._validate_single_citation(
                citation,
                consultation_context
            )
            validation_results.append(result)
        
        # Generate report
        report = ValidationReport(
            status='COMPLETE',
            total_citations=len(citations),
            verified=sum(1 for r in validation_results if r.verified),
            failed=sum(1 for r in validation_results if not r.verified),
            details=validation_results,
            warnings=self._generate_warnings(validation_results)
        )
        
        # Record validation
        self.validation_history.append({
            'timestamp': datetime.utcnow(),
            'report': report
        })
        
        return report
    
    async def _validate_single_citation(
        self,
        citation: Citation,
        context: ConsultationContext
    ) -> CitationValidationResult:
        """
        Validate a single citation.
        """
        
        # Load source document
        doc_content = await self._load_document(citation.document_id)
        
        if not doc_content:
            return CitationValidationResult(
                citation=citation,
                verified=False,
                failure_reason='DOCUMENT_NOT_FOUND',
                confidence=0.0
            )
        
        # Verify section exists
        section_content = self._extract_section(
            doc_content,
            citation.section_id
        )
        
        if not section_content:
            return CitationValidationResult(
                citation=citation,
                verified=False,
                failure_reason='SECTION_NOT_FOUND',
                confidence=0.0
            )
        
        # Validate content
        if citation.is_direct_quote:
            # For direct quotes, require exact match
            quote_found = self._verify_exact_quote(
                section_content,
                citation.quoted_text
            )
            confidence = 1.0 if quote_found else 0.0
            
        else:
            # For paraphrases, require high semantic similarity
            similarity = await self._compute_semantic_similarity(
                section_content,
                citation.paraphrased_content
            )
            quote_found = similarity >= self.violation_threshold
            confidence = similarity
        
        return CitationValidationResult(
            citation=citation,
            verified=quote_found,
            failure_reason=None if quote_found else 'CONTENT_MISMATCH',
            confidence=confidence,
            section_content_preview=section_content[:200]
        )
    
    def _extract_citations(self, response: str) -> List[Citation]:
        """
        Extract all citation references from response text.
        """
        citations = []
        
        # Pattern 1: Direct citation format [doc-name, Section X.Y]
        pattern1 = r'\[([^,]+),\s*Section\s+([\d.]+)\]'
        matches1 = re.finditer(pattern1, response)
        
        for match in matches1:
            doc_id = match.group(1).strip()
            section_id = match.group(2).strip()
            
            # Extract cited content (context around citation)
            start = max(0, match.start() - 200)
            end = min(len(response), match.end() + 200)
            context = response[start:end]
            
            citations.append(Citation(
                document_id=doc_id,
                section_id=section_id,
                cited_content=context,
                is_direct_quote=self._is_direct_quote(context),
                position=match.start()
            ))
        
        # Pattern 2: Template citation format [template-name, Template ID]
        pattern2 = r'\[([^,]+),\s*Template\s+([^\]]+)\]'
        matches2 = re.finditer(pattern2, response)
        
        for match in matches2:
            doc_id = match.group(1).strip()
            template_id = match.group(2).strip()
            
            start = max(0, match.start() - 200)
            end = min(len(response), match.end() + 200)
            context = response[start:end]
            
            citations.append(Citation(
                document_id=doc_id,
                section_id=template_id,
                cited_content=context,
                is_direct_quote=self._is_direct_quote(context),
                position=match.start()
            ))
        
        return citations
    
    def _generate_warnings(
        self,
        validation_results: List[CitationValidationResult]
    ) -> List[str]:
        """Generate warnings for validation issues."""
        
        warnings = []
        
        failed = [r for r in validation_results if not r.verified]
        
        if failed:
            warnings.append(
                f"CRITICAL: {len(failed)} citation(s) failed verification"
            )
            
            for result in failed:
                if result.failure_reason == 'DOCUMENT_NOT_FOUND':
                    warnings.append(
                        f"Document '{result.citation.document_id}' not found in project knowledge"
                    )
                elif result.failure_reason == 'SECTION_NOT_FOUND':
                    warnings.append(
                        f"Section '{result.citation.section_id}' not found in '{result.citation.document_id}'"
                    )
                elif result.failure_reason == 'CONTENT_MISMATCH':
                    warnings.append(
                        f"Cited content does not match source (similarity: {result.confidence:.2f})"
                    )
        
        low_confidence = [r for r in validation_results if r.verified and r.confidence < 0.90]
        if low_confidence:
            warnings.append(
                f"WARNING: {len(low_confidence)} citation(s) have low confidence (<0.90)"
            )
        
        return warnings
```

```python
class KnowledgeVersionTracker:
    """
    Track project knowledge versions and detect changes.
    
    NEW in v5.1.0: Ensures cache freshness and knowledge currency.
    """
    
    def __init__(self):
        self.version_manifest = {}
        self.change_log = []
        self.last_check = None
    
    async def check_for_changes(self) -> ChangeSet:
        """
        Check all project knowledge documents for changes.
        """
        
        current_manifest = await self._build_current_manifest()
        previous_manifest = self._load_previous_manifest()
        
        changes = ChangeSet()
        
        # Detect new documents
        for doc_id in current_manifest:
            if doc_id not in previous_manifest:
                changes.add_new_document(
                    doc_id,
                    current_manifest[doc_id]
                )
        
        # Detect modified documents
        for doc_id in current_manifest:
            if doc_id in previous_manifest:
                if self._has_changed(
                    current_manifest[doc_id],
                    previous_manifest[doc_id]
                ):
                    changes.add_modified_document(
                        doc_id,
                        previous_manifest[doc_id],
                        current_manifest[doc_id]
                    )
        
        # Detect deleted documents
        for doc_id in previous_manifest:
            if doc_id not in current_manifest:
                changes.add_deleted_document(
                    doc_id,
                    previous_manifest[doc_id]
                )
        
        # Save current manifest
        await self._save_manifest(current_manifest)
        
        # Record change check
        self.last_check = datetime.utcnow()
        
        return changes
    
    async def handle_changes(self, changes: ChangeSet):
        """
        Handle detected changes with appropriate actions.
        """
        
        if not changes.has_changes():
            return
        
        severity = changes.assess_severity()
        
        if severity == 'CRITICAL':
            # Critical changes: invalidate all caches, rebuild indices
            await self._handle_critical_changes(changes)
            
        elif severity == 'MODERATE':
            # Moderate changes: selective invalidation
            await self._handle_moderate_changes(changes)
            
        else:
            # Minor changes: incremental update
            await self._handle_minor_changes(changes)
        
        # Log changes
        self.change_log.append({
            'timestamp': datetime.utcnow(),
            'changes': changes,
            'severity': severity,
            'action_taken': self._describe_action(severity)
        })
    
    async def _build_current_manifest(self) -> Dict[str, DocVersion]:
        """
        Build manifest of current project knowledge state.
        """
        
        manifest = {}
        
        # Get all project knowledge documents
        documents = await self._list_project_documents()
        
        for doc in documents:
            # Compute content hash
            content = await self._load_document_content(doc.id)
            content_hash = hashlib.sha256(
                content.encode()
            ).hexdigest()
            
            # Get metadata
            metadata = await self._load_document_metadata(doc.id)
            
            manifest[doc.id] = DocVersion(
                document_id=doc.id,
                content_hash=content_hash,
                last_modified=metadata.get('last_modified'),
                version=metadata.get('version', '1.0.0'),
                size_bytes=len(content)
            )
        
        return manifest
    
    async def _handle_critical_changes(self, changes: ChangeSet):
        """
        Handle critical changes with full invalidation.
        """
        
        # Invalidate all caches
        await cache_manager.invalidate_all()
        
        # Rebuild pattern index
        await pattern_index.rebuild()
        
        # Notify system
        await notification_system.send_alert(
            severity='CRITICAL',
            message=f"Critical knowledge changes detected: {changes.summary()}",
            action='Full cache and index rebuild initiated'
        )
    
    async def _handle_moderate_changes(self, changes: ChangeSet):
        """
        Handle moderate changes with selective invalidation.
        """
        
        # Identify affected cache entries
        affected_queries = await self._identify_affected_queries(changes)
        
        # Invalidate affected caches
        for query in affected_queries:
            await cache_manager.invalidate(query)
        
        # Update pattern index incrementally
        for modified_doc in changes.modified_documents:
            await pattern_index.update_document(modified_doc)
        
        # Notify system
        await notification_system.send_alert(
            severity='MODERATE',
            message=f"Moderate knowledge changes detected: {len(affected_queries)} cache entries invalidated",
            action='Selective invalidation and incremental update completed'
        )
```

```python
            'validate prompt',

        ],
        
        'integration_topics': [
            'combine techniques',
            'integrate with',
            'use together',
            'workflow for',
            'pipeline for',
            'architecture for'
        ],
        
        'production_topics': [
            'deploy prompt',
            'production use',
            'scale prompt',
            'monitor prompt',
            'cost optimization'
        ]
    }
    
    CONTEXT_TRIGGERS = {
        'document_references': [
            'doc1', 'doc2', 'doc3', 'doc4', 'doc5', 'doc6',
            'reasoning techniques guide',
            'extended thinking guide',
            'integration patterns'
        ],
        
        'framework_mentions': [
            'thinking framework',
            'reasoning framework',
            'validation framework',
            'evaluation framework'
        ],
        
        'comparative_queries': [
            'difference between',
            'compare',
            'versus',
            'vs',
            'when to use',
            'which is better'
        ]
    }
    
    def should_search(self, query: str) -> Dict[str, Any]:
        """Determine if project knowledge search is required."""
        query_lower = query.lower()
        
        # Check mandatory triggers
        mandatory_matches = []
        for category, patterns in self.MANDATORY_TRIGGERS.items():
            matches = [p for p in patterns if p in query_lower]
            if matches:
                mandatory_matches.append({
                    'category': category,
                    'matched_patterns': matches
                })
        
        # Check context triggers
        context_matches = []
        for category, patterns in self.CONTEXT_TRIGGERS.items():
            matches = [p for p in patterns if p in query_lower]
            if matches:
                context_matches.append({
                    'category': category,
                    'matched_patterns': matches
                })
        
        requires_search = len(mandatory_matches) > 0 or len(context_matches) > 0
        
        return {
            'requires_search': requires_search,
            'confidence': 'MANDATORY' if mandatory_matches else 'RECOMMENDED',
            'mandatory_matches': mandatory_matches,
            'context_matches': context_matches,
            'suggested_queries': self._generate_search_queries(
                mandatory_matches, context_matches
            )
        }
    
    def _generate_search_queries(
        self, 
        mandatory_matches: List[Dict],
        context_matches: List[Dict]
    ) -> List[str]:
        """Generate specific search queries based on triggers."""
        queries = []
        
        # Generate from mandatory matches
        for match in mandatory_matches:
            for pattern in match['matched_patterns']:
                queries.append(pattern)
                queries.append(f"{pattern} implementation")
                queries.append(f"{pattern} best practices")
        
        # Generate from context matches
        for match in context_matches:
            for pattern in match['matched_patterns']:
                queries.append(pattern)
        
        # Deduplicate and return top 5
        return list(dict.fromkeys(queries))[:5]
```

**Following decision tree from [doc2, Section 3.5]:**

```python
class ThinkingModeSelector:
    """
    Systematic selection of optimal thinking mode.
    
    Implementation follows:
    - doc2-extended-thinking-architecture-implementation-guide.md, Section 3.5
    - doc1-llm-reasoning-techniques-operational-manual.md, Section 4.2
    """
    
    def select_mode(self, task_characteristics: Dict[str, Any]) -> Tuple[str, str]:
        """
        Decision tree for mode selection (per doc2).
        
        Returns:
            Tuple of (mode_name, reasoning)
        """
        
        # Critical factor 1: Latency requirement (doc2, Section 3.5.1)
        if task_characteristics.get('latency_critical'):
            return 'disabled', "Real-time requirement demands no thinking overhead"
        
        # Critical factor 2: Tool use required (doc4, Section 2.1)
        if task_characteristics.get('requires_tools'):
            return 'interleaved', "Tool use benefits from reasoning between actions"
        
        # Factor 3: Complexity assessment (doc1, Section 4.2)
        complexity = self._assess_complexity(task_characteristics)
        
        if complexity == 'simple':
            if task_characteristics.get('token_constrained'):
                return 'disabled', "Simple task + cost constraint = no thinking needed"
            else:
                return 'auto', "Simple task, let Claude decide"
        
        elif complexity == 'moderate':
            if task_characteristics.get('quality_critical'):
                return 'enabled', "Moderate complexity + quality focus = always think"
            else:
                return 'auto', "Moderate complexity, adaptive approach"
        
        else:  # high complexity
            return 'enabled', "High complexity always benefits from extended thinking"
    
    def _assess_complexity(self, characteristics: Dict[str, Any]) -> str:
        """Quantify task complexity per doc1, Section 4.2.1."""
        score = 0
        
        if characteristics.get('multi_step'): score += 2
        if characteristics.get('ambiguous'): score += 2
        if characteristics.get('creative'): score += 1
        if characteristics.get('specialized_domain'): score += 1
        if characteristics.get('multiple_valid_solutions'): score += 2
        
        if score <= 2: return 'simple'
        elif score <= 5: return 'moderate'
        else: return 'high'
```
