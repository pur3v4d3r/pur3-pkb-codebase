>     """Add two numbers"""

>

> def get_version() -> str:
>     """Get server version"""

>

>     }

> 
> # Run server

>     mcp.run(transport="stdio")  # Local
>     # mcp.run(transport="http", host="127.0.0.1", port=8000)  # Remote
